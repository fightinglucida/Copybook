{{- /* Only article-specific images; no preset library fallback. */ -}} {{- $thumb := "" -}} {{- $source := .Params.cover | default .Params.thumbnail -}} {{- with $source -}} {{- $url := urls.Parse . -}} {{- if $url.IsAbs -}}{{- $thumb = . -}} {{- else -}} {{- with $.Resources.GetMatch $url.Path -}}{{- $thumb = .RelPermalink -}} {{- else -}}{{- $thumb = $source | relURL -}}{{- end -}} {{- end -}} {{- end -}} {{- if not $thumb -}} {{- with .Resources.GetMatch "{cover,thumbnail,featured}*" -}}{{- $thumb = .RelPermalink -}}{{- end -}} {{- end -}} {{- return $thumb -}}
